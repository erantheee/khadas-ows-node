cmake_minimum_required(VERSION 3.20)
project(khadas_ows_node LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(khadas_ows_core src/audio_core.cpp src/policy.cpp)
target_include_directories(khadas_ows_core PUBLIC include)
target_compile_options(khadas_ows_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(khadas-ows-node src/main.cpp)
target_link_libraries(khadas-ows-node PRIVATE khadas_ows_core)
enable_testing()
add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE khadas_ows_core)
add_test(NAME core_test COMMAND core_test)
add_executable(policy_test tests/policy_test.cpp)
target_link_libraries(policy_test PRIVATE khadas_ows_core)
add_test(NAME policy_test COMMAND policy_test)
